
int add(int a, int b) {
    int result = a + b;
    return result;
}

int factorial(int n) {
    int i = 1;
    int f = 1;

    while(i < n) {
        f = f * i;
        i = i + 1;
    }

    return f;
}

int max3(int a, int b, int c) {
    int m = a;

    if(b > m) {
        m = b;
    }
    if(c > m) {
        m = c;
    }

    return m;
}


int computeNested(int x) {
    int y = x + 10;
    int z = y * 2;

    if(z > 50) {
        z = z - 5;

        if(z > 80) {
            z = z / 2;
        }
    }
    else {
        z = z + 100;
    }

    return z;
}
